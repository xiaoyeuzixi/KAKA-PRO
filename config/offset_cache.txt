DWORD Decrypt::CIndex(DWORD value)
{
    return ((((value ^ 0x7360F24) << 25) | (((value ^ 0x7360F24) >> 7) & 0x1FF0000)) ^ _rotr(value ^ 0x7360F24, 23) ^ 0xB621EC05);
}
